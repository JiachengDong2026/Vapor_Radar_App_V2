cmake_minimum_required(VERSION 3.21)
project(VaporMonitor VERSION 0.2.5 LANGUAGES CXX)
get_filename_component(VAPOR_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(VAPOR_BINARY_ROOT "${CMAKE_BINARY_DIR}" REALPATH)
string(TOLOWER "${VAPOR_REPO_ROOT}/" VAPOR_REPO_PREFIX)
string(TOLOWER "${VAPOR_BINARY_ROOT}/" VAPOR_BINARY_PREFIX)
string(FIND "${VAPOR_BINARY_PREFIX}" "${VAPOR_REPO_PREFIX}" VAPOR_IN_REPO)
if(VAPOR_IN_REPO EQUAL 0)
 message(FATAL_ERROR "Build outside the formal repository; use Vapor_Lc_App_Test.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network)
add_executable(VaporMonitor WIN32 src/main.cpp src/mainwindow.cpp src/traceplot.cpp src/worker.cpp src/protocol.cpp)
target_include_directories(VaporMonitor PRIVATE src)
target_link_libraries(VaporMonitor PRIVATE Qt6::Widgets Qt6::Network)
add_executable(control_smoke tests/control_smoke.cpp src/worker.cpp src/protocol.cpp)
target_include_directories(control_smoke PRIVATE src)
target_link_libraries(control_smoke PRIVATE Qt6::Network)
add_executable(ui_test tests/ui_test.cpp src/mainwindow.cpp src/traceplot.cpp)
target_include_directories(ui_test PRIVATE src)
target_link_libraries(ui_test PRIVATE Qt6::Widgets)
if(MINGW)
 target_compile_options(VaporMonitor PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_test.cpp")
 add_executable(protocol_test tests/protocol_test.cpp src/protocol.cpp)
 target_include_directories(protocol_test PRIVATE src)
 target_link_libraries(protocol_test PRIVATE Qt6::Core)
 enable_testing()
 add_test(NAME protocol COMMAND protocol_test)
endif()

add_executable(plot_test tests/plot_test.cpp src/traceplot.cpp)
target_include_directories(plot_test PRIVATE src)
target_link_libraries(plot_test PRIVATE Qt6::Widgets)
add_test(NAME ui COMMAND ui_test)
add_test(NAME plot COMMAND plot_test)
set_tests_properties(ui plot PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
